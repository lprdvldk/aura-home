cmake_minimum_required(VERSION 3.24)
if(NOT CMAKE_CXX_COMPILER)
  find_program(SMART_HOUSE_GXX g++)
  if(SMART_HOUSE_GXX)
    set(CMAKE_CXX_COMPILER "${SMART_HOUSE_GXX}" CACHE FILEPATH "C++ compiler" FORCE)
  endif()
endif()
project(smart_house VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

include(CTest)

add_subdirectory(libs/dht11)
add_subdirectory(apps/dht11-agent)

option(SMART_HOUSE_BUILD_DESKTOP "Build the Qt6 desktop app" ON)
if(SMART_HOUSE_BUILD_DESKTOP)
  find_package(Qt6 QUIET COMPONENTS Widgets Charts WebSockets Network)
  if(Qt6_FOUND)
    add_subdirectory(apps/desktop)
  else()
    message(STATUS "Qt6 not found — skipping apps/desktop. Install Qt 6.5+ and reconfigure.")
  endif()
endif()
